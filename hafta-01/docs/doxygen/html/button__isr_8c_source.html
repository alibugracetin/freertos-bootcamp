65void button_rearm_poll(void)
66{
67 static bool s_low_seen; /* son çağrıdan beri kesintisiz LOW mu */
68 static uint32_t s_low_since_us; /* ilk LOW gözleminin zamanı */
69
70 if (s_armed) {
71 s_low_seen = false;
72 return;
73 }
74
75 const uint32_t now = timer_us();
76
77 if (HAL_GPIO_ReadPin(BTN_USER_GPIO_Port, BTN_USER_Pin) != GPIO_PIN_RESET) {
78 s_low_seen = false; /* hâlâ basılı veya zıplıyor: süre baştan */
79 return;
80 }
81 if (!s_low_seen) {
82 s_low_seen = true;
83 s_low_since_us = now;
84 return;
85 }
86 if ((uint32_t)(now - s_low_since_us) < APP_REARM_LOW_US) {
87 return;
88 }
89
90 /* Pin okuma ile bayrak yazma arasına bir basış kenarı girerse: EXTI bekleyen
91 * kesme olarak kilitlenir, kritik bölümden çıkınca silah açık bulunur ve
92 * kabul edilir. Bölüm yalnızca birkaç komut sürer. */
93 taskENTER_CRITICAL();
94 if (HAL_GPIO_ReadPin(BTN_USER_GPIO_Port, BTN_USER_Pin) == GPIO_PIN_RESET) {
95 s_armed = true;
96 }
97 taskEXIT_CRITICAL();
98 s_low_seen = false;
99}

124void HAL_GPIO_EXTI_Callback(uint16_t GPIO_Pin)
125{
126 if (GPIO_Pin != BTN_USER_Pin || g_button_q == NULL) {
127 return;
128 }
129
130 const uint32_t t0 = s_entry_t0;
131
132 if (s_have_accept && (uint32_t)(t0 - s_last_accept_us) < APP_DEBOUNCE_US) {
133 g_cnt_isr.debounce_rej++; /* FR-12 */
134 return;
135 }
136 if (!s_armed) {
137 g_cnt_isr.unarmed_rej++; /* FR-12b */
138 return;
139 }
140 s_armed = false;
141 s_have_accept = true;
142 s_last_accept_us = t0;
143
144 if (!exp_is_measuring()) {
145 g_cnt_isr.idle_press++; /* FR-87: kayda alınmaz */
146 return;
147 }
148
149 const uint32_t id = ++s_event_counter; /* FR-14 */
150 g_cnt_isr.events++;
151
152 if (!rec_open(id, t0)) {
153 g_cnt_isr.rec_ovf++; /* FR-61 */
154 return;
155 }
156
157 const ButtonEvent e = { .event_id = id, .t0_us = t0 };
158 BaseType_t woken = pdFALSE;
159
160 if (xQueueSendFromISR(g_button_q, &e, &woken) != pdPASS) { /* FR-15 */
161 g_cnt_isr.btnq_drop++;
162 (void)rec_close(id, REC_BTNQ_DROP); /* FR-18: kimlik ve t₀ korunur */
163 } else {
164 g_cnt_isr.accepted++;
165 }
166
167 /* ButtonTask (öncelik 2) şu an koşan görevden yüksekse, kesmeden çıkar
168 * çıkmaz ona geçilir; bir sonraki tick beklenmez. */
169 portYIELD_FROM_ISR(woken);
170}
