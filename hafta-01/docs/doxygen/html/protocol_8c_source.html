54bool proto_parse_cmd(const char *line, Command *out)
55{
56 out->type = CMD_INVALID;
57 out->scenario = 0u;
58 out->count = 0u;
59
60 if (strncmp(line, "CMD,", 4) != 0) {
61 return false;
62 }
63 const char *arg = line + 4;
64
65 if (strcmp(arg, "STOP") == 0) { out->type = CMD_STOP; return true; }
66 if (strcmp(arg, "DUMP") == 0) { out->type = CMD_DUMP; return true; }
67 if (strcmp(arg, "STAT") == 0) { out->type = CMD_STAT; return true; }
68
69 if (strncmp(arg, "SCEN,S", 6) == 0 && arg[6] >= '0' && arg[6] <= '5' && arg[7] == '\0') {
70 out->type = CMD_SCEN;
71 out->scenario = (uint8_t)(arg[6] - '0');
72 return true;
73 }
74
75 if (strncmp(arg, "START", 5) == 0) {
76 if (arg[5] == '\0') {
77 out->type = CMD_START;
78 return true;
79 }
80 if (arg[5] == ',') {
81 char *end = NULL;
82 const unsigned long n = strtoul(&arg[6], &end, 10);
83 if (end != &arg[6] && *end == '\0' && n >= 1u && n <= APP_REC_POOL_SIZE) {
84 out->type = CMD_START;
85 out->count = (uint32_t)n;
86 return true;
87 }
88 }
89 }
90 return false;
91}
