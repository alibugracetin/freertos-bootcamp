61static void st_printf(const char *fmt, ...)
62{
63 char buf[128];
64 va_list ap;
65
66 va_start(ap, fmt);
67 int n = vsnprintf(buf, sizeof buf, fmt, ap);
68 va_end(ap);
69
70 if (n < 0) {
71 return;
72 }
73 if ((size_t)n >= sizeof buf) {
74 n = (int)sizeof buf - 1; /* tanı çıktısında kesme kabul edilebilir */
75 }
76 (void)HAL_UART_Transmit(&huart2, (uint8_t *)buf, (uint16_t)n, 100u);
77}

96static void st_led_chase(void)
97{
98 static const struct { GPIO_TypeDef *port; uint16_t pin; const char *name; } leds[] = {
99 { LED_GREEN_GPIO_Port, LED_GREEN_Pin, "GREEN" },
100 { LED_ORANGE_GPIO_Port, LED_ORANGE_Pin, "ORANGE" },

107 st_printf("HWTEST,LED,%s\n", leds[i].name);
108 HAL_Delay(300u);
109 HAL_GPIO_WritePin(leds[i].port, leds[i].pin, GPIO_PIN_RESET);
110 }

127static void st_timer_check(void)
128{
129 const uint32_t hclk_mhz = HAL_RCC_GetHCLKFreq() / 1000000u;
130
131 for (uint32_t r = 1; r <= ST_TIM_ROUNDS; r++) {
132 uint32_t t0 = st_timer_us();
133 uint32_t c0 = DWT->CYCCNT;
134 HAL_Delay(1000u);
135 uint32_t t1 = st_timer_us();
136 uint32_t c1 = DWT->CYCCNT;
137
138 uint32_t tim_us = t1 - t0; /* mod 2^32, FR-68 */
139 uint32_t dwt_us = (c1 - c0) / hclk_mhz;
140 st_printf("HWTEST,TIM,round=%lu,tim2_us=%lu,dwt_us=%lu,diff_us=%ld\n",
141 (unsigned long)r, (unsigned long)tim_us, (unsigned long)dwt_us,
142 (long)((int32_t)(tim_us - dwt_us)));
143 }
144
145 /* İşaretsiz çıkarmanın sarmayı doğru işlediğinin sayısal kanıtı (FR-68):
146 * sayaç 0xFFFFFFF0'da başlayıp 0x00000010'a sarmış olsun → gerçek aralık 32 µs. */
147 const uint32_t before = 0xFFFFFFF0u;
148 const uint32_t after = 0x00000010u;
149 st_printf("HWTEST,WRAP,before=0x%08lX,after=0x%08lX,delta=%lu,expected=32\n",
150 (unsigned long)before, (unsigned long)after, (unsigned long)(after - before));
151}

168void hw_selftest_run(void)
169{
170 st_dwt_init();
171 (void)HAL_TIM_Base_Start(&htim2); /* TIM2 başlatılmadan CNT sabit kalır */
172
173 st_printf("\nHWTEST,BOOT,SYSCLK=%lu,HCLK=%lu,PCLK1=%lu,PCLK2=%lu\n",
174 (unsigned long)HAL_RCC_GetSysClockFreq(), (unsigned long)HAL_RCC_GetHCLKFreq(),
175 (unsigned long)HAL_RCC_GetPCLK1Freq(), (unsigned long)HAL_RCC_GetPCLK2Freq());
176
177 st_led_chase();
178 st_timer_check();
179
180 st_printf("HWTEST,INFO,btn_level_idle=%u,press the blue button\n",
181 (unsigned)HAL_GPIO_ReadPin(BTN_USER_GPIO_Port, BTN_USER_Pin));
182
183 GPIO_PinState last_level = HAL_GPIO_ReadPin(BTN_USER_GPIO_Port, BTN_USER_Pin);
184 /* Kenarlar bir bırakıştan sonrakine kadar sayılır. Anlık görüntüyü basışta
185 * almak yanlış olurdu: yoklama basışı ≤5 ms geç fark eder ve kesme o arada
186 * ilk kenarı ve zıplamaların bir kısmını çoktan saymış olur. */
187 uint32_t edges_at_release = s_exti_edges;
188 uint32_t presses = 0;
189 uint32_t alive = 0;
190 uint32_t last_alive = HAL_GetTick();
191
192 for (;;) {
193 GPIO_PinState level = HAL_GPIO_ReadPin(BTN_USER_GPIO_Port, BTN_USER_Pin);
194
195 if (level != last_level) {
196 if (level == GPIO_PIN_SET) {
197 presses++;
198 HAL_GPIO_WritePin(LED_GREEN_GPIO_Port, LED_GREEN_Pin, GPIO_PIN_SET);
199 st_printf("HWTEST,BTN,press=%lu,level=1,t_us=%lu\n",
200 (unsigned long)presses, (unsigned long)st_timer_us());
201 } else {
202 uint32_t now_edges = s_exti_edges;
203 HAL_GPIO_WritePin(LED_GREEN_GPIO_Port, LED_GREEN_Pin, GPIO_PIN_RESET);
204 /* 1 = temiz basış; >1 = zıplama (bounce) var → FR-12 filtresi gerekli */
205 st_printf("HWTEST,BTN,release=%lu,level=0,rising_edges_this_cycle=%lu\n",
206 (unsigned long)presses, (unsigned long)(now_edges - edges_at_release));
207 edges_at_release = now_edges;
208 }
209 last_level = level;
210 }
211
212 if (HAL_GetTick() - last_alive >= ST_ALIVE_PERIOD_MS) {
213 /* Turuncu → kırmızı → mavi sürekli döner; kullanıcı LED'leri kendi
214 * hızında gözle doğrulayabilir. Yeşil butona ayrılmıştır. */
215 static const struct { uint16_t pin; const char *name; } ring[] = {
216 { LED_ORANGE_Pin, "ORANGE" }, { LED_RED_Pin, "RED" }, { LED_BLUE_Pin, "BLUE" },
217 };
218 const unsigned k = alive % (sizeof ring / sizeof ring[0]);
219
220 last_alive += ST_ALIVE_PERIOD_MS;
221 HAL_GPIO_WritePin(GPIOD, LED_ORANGE_Pin | LED_RED_Pin | LED_BLUE_Pin, GPIO_PIN_RESET);
222 HAL_GPIO_WritePin(GPIOD, ring[k].pin, GPIO_PIN_SET);
223 st_printf("HWTEST,ALIVE,n=%lu,led=%s,t_us=%lu,exti_edges=%lu\n",
224 (unsigned long)++alive, ring[k].name, (unsigned long)st_timer_us(),
225 (unsigned long)s_exti_edges);
226 }
227
228 HAL_Delay(ST_BTN_POLL_MS);
229 }
230}
