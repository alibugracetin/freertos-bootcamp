150static void set_leds(ExpState st)
151{
152 HAL_GPIO_WritePin(GPIOD, LED_GREEN_Pin | LED_ORANGE_Pin | LED_RED_Pin | LED_BLUE_Pin, GPIO_PIN_RESET);

154 switch (st) {

157 case EXP_DRAINING:
158 case EXP_DONE: pin = LED_RED_Pin; break;

243static void dump_records(void)
244{
245 const char *sn = k_scen[s_scen].name;
246 uint32_t n = 0u;
247
248 for (uint32_t s = 0; s < APP_REC_POOL_SIZE; s++) {
249 EventRecord r;
250 if (!rec_read(s, &r) || r.status == REC_FREE) {
251 continue;
252 }
253 char f[TS_COUNT][11];
254 for (unsigned i = 0; i < TS_COUNT; i++) {
255 if (r.have & (1u << i)) {
256 (void)snprintf(f[i], sizeof f[i], "%lu", (unsigned long)r.t[i]);
257 } else {
258 f[i][0] = '\0';
259 }
260 }
261 TxMsg m;
262 if (proto_fmt_line(&m, MSG_REC, "REC,%s,%lu,%s,%s,%s,%s,%s,%s", sn, (unsigned long)r.event_id,
263 f[0], f[1], f[2], f[3], f[4], status_name(r.status))) {
264 send_ctrl(&m);
265 n++;
266 }
267 }
268
269 const uint32_t per_avg = s_tel.per_n ? s_tel.per_sum / s_tel.per_n : 0u;
270 const uint32_t work_avg = s_tel.work_n ? s_tel.work_sum / s_tel.work_n : 0u;
271 SEND_LINE("TST,%s,%lu,%lu,%lu,%lu,%lu,%lu", sn, (unsigned long)s_tel.sent,
272 (unsigned long)(s_tel.per_n ? s_tel.per_min : 0u), (unsigned long)per_avg,
273 (unsigned long)s_tel.per_max, (unsigned long)work_avg, (unsigned long)s_tel.work_max);
274
275 const volatile IsrCounters *ci = &g_cnt_isr;
276 SEND_LINE("CNI,%lu,%lu,%lu,%lu,%lu,%lu,%lu,%lu,%lu,%lu,%lu",
277 (unsigned long)ci->accepted, (unsigned long)ci->debounce_rej, (unsigned long)ci->btnq_drop,
278 (unsigned long)ci->rec_ovf, (unsigned long)ci->rec_mismatch, (unsigned long)ci->unarmed_rej,
279 (unsigned long)ci->events, (unsigned long)ci->idle_press, (unsigned long)ci->rx_overflow,
280 (unsigned long)ci->rx_err, (unsigned long)ci->cmd_busy);
281
282 const volatile TaskCounters *ct = &g_cnt_task;
283 SEND_LINE("CNT,%lu,%lu,%lu,%lu,%lu,%lu,%lu,%lu,%lu,%lu",
284 (unsigned long)ct->txq_drop_tel, (unsigned long)ct->txq_drop_btn, (unsigned long)ct->rec_mismatch,
285 (unsigned long)ct->uart_err, (unsigned long)ct->uart_timeout, (unsigned long)ct->rec_mismatch_tx,
286 (unsigned long)ct->fmt_err_tel, (unsigned long)ct->fmt_err_btn, (unsigned long)s_txq_hwm_meas,
287 (unsigned long)ct->cmd_err);
288
289 SEND_LINE("END,%s,%lu", sn, (unsigned long)n);
290}

310static void handle_command(const char *line)
311{
312 Command c;
313
314 if (!proto_parse_cmd(line, &c)) {
315 reject("unknown", line);
316 return;
317 }
318
319 switch (c.type) {
320 case CMD_SCEN:
321 if (s_state != EXP_IDLE && s_state != EXP_ARMED && s_state != EXP_DONE) {
322 reject("state", line); /* FR-85 */
323 return;
324 }
325 if (s_tel_active) {
326 /* STOP'tan sonra telemetri en fazla bir periyot (≤100 ms) daha döner;
327 * o bitmeden istatistikler sıfırlanırsa yarışırdı. */
328 reject("busy", line);
329 return;
330 }
331 s_scen = c.scenario;
332 reset_measurement();
333 enter(EXP_ARMED);
334 SEND_LINE("ACK,SCEN,%s,%u,%lu", k_scen[s_scen].name, (unsigned)k_scen[s_scen].period_ms,
335 (unsigned long)k_scen[s_scen].work_iters);
336 send_status();
337 break;
338
339 case CMD_START:
340 if (s_state != EXP_ARMED) {
341 reject("state", line);
342 return;
343 }
344 s_target = c.count ? c.count : APP_DEFAULT_TARGET;
345 enter(EXP_WARMUP);
346 SEND_LINE("ACK,START,%lu", (unsigned long)s_target);
347 send_status();
348 telemetry_start(); /* ısınmada telemetri koşar (FR-87) */
349 break;
350
351 case CMD_STOP:
352 s_measuring = false;
353 telemetry_stop();
354 enter(EXP_IDLE);
355 SEND_LINE("ACK,STOP");
356 send_status();
357 break;
358
359 case CMD_DUMP:
360 if (s_state != EXP_DONE) {
361 reject("state", line);
362 return;
363 }
364 dump_records();
365 break;
366
367 case CMD_STAT:
368 if (s_state == EXP_MEASURING) {
369 reject("state", line); /* FR-91: ölçümde ek trafik yok */
370 return;
371 }
372 send_status();
373 break;
374
375 default:
376 reject("unknown", line);
377 break;
378 }
379}

383void exp_tick(void)
384{
385 char line[32];
386
387 if (!s_announced) { /* açılışta bir kez */
388 s_announced = true;
389 s_tel.per_min = UINT32_MAX;
390 enter(EXP_IDLE);
391 SEND_LINE("BOOT,hafta01,work_us_per_100k=%lu", (unsigned long)g_diag.work_us_per_100k);
392 send_status();
393 }
394
395 if (uart_link_take_command(line, sizeof line)) {
396 handle_command(line);
397 }
398
399 const TickType_t elapsed = xTaskGetTickCount() - s_state_since;
400
401 switch (s_state) {
402 case EXP_WARMUP:
403 if (elapsed >= pdMS_TO_TICKS(APP_WARMUP_MS)) {
404 enter(EXP_MEASURING);
405 send_status(); /* ölçüm başlamadan önceki son kontrol mesajı */
406 s_measuring = true;
407 }
408 break;
409
410 case EXP_MEASURING:
411 if (g_cnt_isr.events >= s_target) {
412 s_measuring = false; /* yeni olay açılmaz */
413 s_txq_hwm_meas = g_cnt_task.txq_hwm; /* dökümden önceki değer */
414 telemetry_stop();
415 enter(EXP_DRAINING);
416 }
417 break;
418
419 case EXP_DRAINING: {
420 const bool drained = !s_tel_active && uxQueueMessagesWaiting(g_tx_q) == 0u &&
421 !uart_link_busy() && rec_count_open() == 0u;
422 if (drained || elapsed >= pdMS_TO_TICKS(APP_DRAIN_TIMEOUT_MS)) {
423 (void)rec_timeout_open(); /* MR-06 */
424 enter(EXP_DONE);
425 send_status();
426 }
427 break;
428 }
429
430 default:
431 break;
432 }
433}

437void TelemetryTask(void *arg)
438{
439 (void)arg;
440 s_tel_task = xTaskGetCurrentTaskHandle();
441 uint32_t seq = 0u;
442
443 for (;;) {
444 (void)ulTaskNotifyTake(pdTRUE, portMAX_DELAY); /* S0 ve IDLE: süresiz bloklu */
445
446 const Scenario *sc = &k_scen[s_scen]; /* koşarken değişmez (FR-85) */
447 const TickType_t per = pdMS_TO_TICKS(sc->period_ms);
448 TickType_t last = xTaskGetTickCount();
449 uint32_t prev_us = 0u;
450 bool have_prev = false;
451
452 s_tel_active = true;
453 while (s_tel_run) {
454 const uint32_t t_start = timer_us();
455
456 if (have_prev) { /* gerçekleşen periyot (AR-11) */
457 const uint32_t p = t_start - prev_us;
458 s_tel.per_n++;
459 s_tel.per_sum += p;
460 if (p < s_tel.per_min) { s_tel.per_min = p; }
461 if (p > s_tel.per_max) { s_tel.per_max = p; }
462 }
463 prev_us = t_start;
464 have_prev = true;
465
466 if (sc->work_iters != 0u) { /* S4 / S5 ek CPU yükü */
467 calibrated_work(sc->work_iters);
468 const uint32_t w = timer_us() - t_start;
469 s_tel.work_n++;
470 s_tel.work_sum += w;
471 if (w > s_tel.work_max) { s_tel.work_max = w; }
472 }
473
474 TxMsg m;
475 if (!proto_fmt_fixed(&m, MSG_TEL, 0u, "TEL,%lu,%s,%lu",
476 (unsigned long)++seq, sc->name, (unsigned long)t_start)) {
477 g_cnt_task.fmt_err_tel++;
478 } else if (xQueueSend(g_tx_q, &m, 0) != pdPASS) { /* FR-26: bloklamaz */
479 g_cnt_task.txq_drop_tel++;
480 }
481 s_tel.sent++;
482
483 vTaskDelayUntil(&last, per); /* FR-20: mutlak periyot */
484 }
485 s_tel_active = false;
486 }
