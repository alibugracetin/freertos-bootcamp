77void UartTxTask(void *arg)
78{
79 (void)arg;
80 s_tx_task = xTaskGetCurrentTaskHandle();
81
82 for (;;) {
83 TxMsg m;
84
85 /* Kuyruk derinliği yalnızca bu görev tarafından azaltılır; alımdan hemen
86 * önceki derinlik, önceki alımdan beri görülen en yüksek değerdir (±1). */
87 const UBaseType_t depth = uxQueueMessagesWaiting(g_tx_q);
88 if (depth > g_cnt_task.txq_hwm) {
89 g_cnt_task.txq_hwm = depth;
90 }
91
92 (void)xQueueReceive(g_tx_q, &m, portMAX_DELAY); /* FR-40: FIFO */
93
94 memcpy(s_tx_buf, m.data, m.len);
95 s_tx_kind = m.kind;
96 s_tx_event = m.event_id;
97 s_tx_busy = true;
98 (void)ulTaskNotifyTake(pdTRUE, 0); /* bayat bildirim kalmasın */
99
100 const uint32_t t3 = timer_us(); /* t₃ — FR-42 */
101 if (m.kind == MSG_BTN) {
102 s_tx_t3 = t3;
103 if (!rec_stamp(m.event_id, TS_T3, t3)) {
104 g_cnt_task.rec_mismatch_tx++;
105 }
106 }
107
108 if (HAL_UART_Transmit_DMA(&huart2, s_tx_buf, m.len) != HAL_OK) {
109 g_cnt_task.uart_err++; /* FR-44 */
110 if (m.kind == MSG_BTN) {
111 (void)rec_close(m.event_id, REC_TX_ERROR);
112 }
113 s_tx_busy = false;
114 continue;
115 }
116
117 if (ulTaskNotifyTake(pdTRUE, pdMS_TO_TICKS(1000)) == 0u) {
118 g_cnt_task.uart_timeout++; /* MR-06: sonsuz bekleme yok */
119 (void)HAL_UART_AbortTransmit(&huart2);
120 if (m.kind == MSG_BTN) {
121 (void)rec_close(m.event_id, REC_TIMEOUT);
122 }
123 }
124 s_tx_busy = false;
125 }
126}

139void HAL_UART_TxCpltCallback(UART_HandleTypeDef *huart)
140{
141 if (huart->Instance != USART2) {
142 return;
143 }
144 const uint32_t t4 = s_uart_entry_us;
145
146 if (s_tx_kind == MSG_BTN) {
147 const uint32_t id = s_tx_event;
148 if (!(rec_stamp(id, TS_T4, t4) && rec_close(id, REC_OK))) {
149 g_cnt_isr.rec_mismatch++;
150 }
151 const uint32_t dma_to_tc = t4 - s_dma_entry_us;
152 g_diag.last_dma_to_tc_us = dma_to_tc;
153 if (dma_to_tc > g_diag.max_dma_to_tc_us) {
154 g_diag.max_dma_to_tc_us = dma_to_tc;
155 }
156 g_diag.last_t4_minus_t3 = t4 - s_tx_t3;
157 }
158
159 if (s_tx_task != NULL) {
160 BaseType_t woken = pdFALSE;
161 vTaskNotifyGiveFromISR(s_tx_task, &woken);
162 portYIELD_FROM_ISR(woken);
163 }
164}

205void HAL_UART_RxCpltCallback(UART_HandleTypeDef *huart)
206{
207 if (huart->Instance != USART2) {
208 return;
209 }
210 const char c = (char)s_rx_byte;
211
212 if (c == '\n') {
213 if (!s_rx_discard) {
214 if (s_cmd_ready) {
215 g_cnt_isr.cmd_busy++;
216 } else {
217 memcpy(s_cmd_line, s_rx_line, s_rx_len);
218 s_cmd_line[s_rx_len] = '\0';
219 s_cmd_ready = true;
220 }
221 }
222 s_rx_len = 0u;
223 s_rx_discard = false;
224 } else if (c != '\r' && !s_rx_discard) {
225 if (s_rx_len < RX_LINE_MAX) {
226 s_rx_line[s_rx_len++] = c;
227 } else {
228 g_cnt_isr.rx_overflow++; /* FR-83: sessiz yutma yok */
229 s_rx_discard = true;
230 }
231 }
232 rx_arm();
233}

249bool uart_link_take_command(char *dst, size_t n)
250{
251 if (s_rx_restart) {
252 s_rx_restart = false;
253 rx_arm();
254 }
255 if (!s_cmd_ready) {
256 return false;
257 }
258 strncpy(dst, s_cmd_line, n - 1u);
259 dst[n - 1u] = '\0';
260 s_cmd_ready = false; /* sahipliği ISR'a geri ver */
261 return true;
262}
