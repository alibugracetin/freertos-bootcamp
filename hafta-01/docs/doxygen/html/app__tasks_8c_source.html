54static void ButtonTask(void *arg)
55{
56 (void)arg;
57 TickType_t last_maint = xTaskGetTickCount();
58
59 for (;;) {
60 ButtonEvent e;
61 const BaseType_t got = xQueueReceive(g_button_q, &e, pdMS_TO_TICKS(APP_BUTTON_POLL_MS));
62
63 if (got == pdPASS) {
64 const uint32_t t1 = timer_us(); /* t₁ — FR-31 */
65 if (!rec_stamp(e.event_id, TS_T1, t1)) {
66 g_cnt_task.rec_mismatch++;
67 }
68
69 TxMsg m;
70 if (!proto_fmt_fixed(&m, MSG_BTN, e.event_id, "BTN,%lu,%s,PRESSED",
71 (unsigned long)e.event_id, exp_scenario_name())) {
72 g_cnt_task.fmt_err_btn++; /* FR-52 */
73 (void)rec_close(e.event_id, REC_TX_ERROR);
74 } else {
75 const uint32_t t2 = timer_us(); /* t₂ — FR-32 */
76 const BaseType_t sent = xQueueSend(g_tx_q, &m, 0);
77 if (!rec_stamp(e.event_id, TS_T2, t2)) {
78 g_cnt_task.rec_mismatch++;
79 }
80 if (sent != pdPASS) {
81 g_cnt_task.txq_drop_btn++; /* FR-34 */
82 (void)rec_close(e.event_id, REC_TXQ_DROP);
83 }
84 }
85
86 const uint32_t lat = t1 - e.t0_us;
87 g_diag.btn_wakeups++;
88 g_diag.last_event_id = e.event_id;
89 g_diag.last_t0 = e.t0_us;
90 g_diag.last_t1 = t1;
91 g_diag.last_t1_minus_t0 = lat;
92 if (lat > g_diag.max_t1_minus_t0) {
93 g_diag.max_t1_minus_t0 = lat;
94 }
95 }
96
97 /* Bakım turu olaydan hemen sonra DEĞİL, olay yokken çalışır (en geç 50 ms'de bir).
98 *
99 * Neden: UartTxTask (öncelik 1) bu görev (öncelik 2) bloklanana kadar koşamaz.
100 * Gönderimden sonra burada yapılan her iş doğrudan t₃ − t₂'ye eklenir. İlk
101 * uçtan uca testte (T-10) bakım turu her olaydan sonra koşuyordu ve boş
102 * kuyrukta bile t₃ − t₂ = 125 µs ölçüldü; bunun büyük kısmı stack'i tarayan
103 * tanı çağrısıydı — yani ölçüm aletinin kendi izi (tasarım §5.2). */
104 const TickType_t now = xTaskGetTickCount();
105 if (got != pdPASS || (now - last_maint) >= pdMS_TO_TICKS(APP_BUTTON_POLL_MS)) {
106 last_maint = now;
107 button_rearm_poll(); /* FR-12b */
108 exp_tick(); /* tasarım §7 */
109
110 g_diag.btn_stack_hwm = (uint32_t)uxTaskGetStackHighWaterMark(NULL);
111 g_diag.task_count = (uint32_t)uxTaskGetNumberOfTasks();
112 g_diag.heap_free = (uint32_t)xPortGetFreeHeapSize();
113 g_diag.heap_min_free = (uint32_t)xPortGetMinimumEverFreeHeapSize();
114 }
115 }
116}

118void app_init_early(void)
119{
120 uint32_t opened = 0u, rejected = 0u;
121
122 timing_init();
123
124 g_diag.magic = APP_DIAG_MAGIC;
125 g_diag.selfcheck = rec_selfcheck(&opened, &rejected) ? 1u : 2u;
126 g_diag.selfcheck_opened = opened;
127 g_diag.selfcheck_rejected = rejected;
128
129 /* Ek CPU işinin bu derlemedeki hızı (T-16 kalibrasyonunun girdisi). Scheduler
130 * öncesi ölçülür; yalnızca HAL tick kesmesi araya girebilir (~1 µs/ms). */
131 const uint32_t t = timer_us();
132 calibrated_work(100000u);
133 g_diag.work_us_per_100k = timer_us() - t;
134
135 uart_link_start_rx(); /* FR-80 */
136}

138void app_create_rtos_objects(void)
139{
140 g_button_q = xQueueCreate(APP_BUTTON_QUEUE_LEN, sizeof(ButtonEvent));
141 g_tx_q = xQueueCreate(APP_TX_QUEUE_LEN, sizeof(TxMsg));
142 configASSERT(g_button_q != NULL && g_tx_q != NULL);
143
144 vQueueAddToRegistry(g_button_q, "buttonQ");
145 vQueueAddToRegistry(g_tx_q, "txQ");
146
147 /* Her dönüş değeri pdPASS ile ayrı ayrı karşılaştırılır. Başarısızlık -1 döner;
148 * `ok &= xTaskCreate(...)` gibi bit düzeyinde birleştirme 1 & -1 = 1 verip
149 * hatayı gizlerdi. */
150 const BaseType_t r_tel = xTaskCreate(TelemetryTask, "Telemetry", APP_TASK_STACK_WORDS,
151 NULL, APP_PRIO_TELEMETRY, NULL);
152 const BaseType_t r_btn = xTaskCreate(ButtonTask, "Button", APP_TASK_STACK_WORDS,
153 NULL, APP_PRIO_BUTTON, NULL);
154 const BaseType_t r_tx = xTaskCreate(UartTxTask, "UartTx", APP_TASK_STACK_WORDS,
155 NULL, APP_PRIO_UARTTX, NULL);
156 configASSERT(r_tel == pdPASS && r_btn == pdPASS && r_tx == pdPASS);
157}
